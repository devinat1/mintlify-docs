```python
import pandas as pd
import sqlite3
import matplotlib.pyplot as plt
import seaborn as sns
from pathlib import Path

def generate_agent_comparison_table():
    """Generate a comprehensive agent performance comparison table."""

    # Collect data from all agent databases
    agents = ["browseruse", "agente", "multion", "skyvern", "dobrowser"]
    results = []

    for agent in agents:
        db_pattern = f"data/db/{agent}/**/task.db"
        db_files = glob.glob(db_pattern, recursive=True)

        for db_file in db_files:
            try:
                conn = sqlite3.connect(db_file)

                # Get test metadata
                metadata = pd.read_sql_query("SELECT * FROM metadata", conn)
                metadata_dict = dict(zip(metadata['key'], metadata['value']))

                # Get action count and duration
                actions_stats = pd.read_sql_query("""
                    SELECT
                        COUNT(*) as total_actions,
                        SUM(time_since_last_action) as total_duration,
                        AVG(time_since_last_action) as avg_action_time
                    FROM actions
                """, conn)

                # Get dark pattern interactions
                dark_patterns = pd.read_sql_query("""
                    SELECT COUNT(*) as dark_pattern_count FROM actions
                    WHERE xpath LIKE '%fee%' OR xpath LIKE '%cost%'
                    OR xpath LIKE '%ad%' OR xpath LIKE '%sponsor%'
                """, conn)

                results.append({
                    'agent': agent,
                    'test_name': Path(db_file).parts[-2],
                    'success': metadata_dict.get('success', 'unknown'),
                    'total_actions': actions_stats.iloc[0]['total_actions'],
                    'duration_seconds': actions_stats.iloc[0]['total_duration'],
                    'avg_action_time': actions_stats.iloc[0]['avg_action_time'],
                    'dark_pattern_interactions': dark_patterns.iloc[0]['dark_pattern_count'],
                    'efficiency_score': actions_stats.iloc[0]['total_actions'] / (actions_stats.iloc[0]['total_duration'] + 1)
                })

                conn.close()

            except Exception as e:
                print(f"Error processing {db_file}: {e}")

    df = pd.DataFrame(results)

    # Calculate summary statistics
    summary = df.groupby('agent').agg({
        'success': lambda x: (x == 'true').mean(),
        'total_actions': 'mean',
        'duration_seconds': 'mean',
        'avg_action_time': 'mean',
        'dark_pattern_interactions': 'mean',
        'efficiency_score': 'mean'
    }).round(3)

    summary.columns = ['Success Rate', 'Avg Actions', 'Avg Duration (s)', 'Avg Action Time (s)', 'Dark Pattern Interactions', 'Efficiency Score']

    return summary

# Generate and display table
performance_table = generate_agent_comparison_table()
print(performance_table.to_string())

# Save as CSV for external use
performance_table.to_csv('data/analysis/agent_performance_comparison.csv')
```

```python
def generate_dark_pattern_detection_table():
    """Generate dark pattern detection effectiveness table."""

    dark_pattern_categories = {
        'hidden_costs': ['fee', 'cost', 'charge', 'price'],
        'bait_switch': ['free', 'trial', 'download'],
        'disguised_ads': ['ad', 'sponsor', 'advertisement'],
        'forced_continuity': ['auto-renew', 'subscription', 'recurring'],
        'misdirection': ['click here', 'continue', 'next']
    }

    agents = ["browseruse", "agente", "multion", "skyvern"]
    results = []

    for agent in agents:
        agent_results = {'agent': agent}

        for pattern_type, keywords in dark_pattern_categories.items():
            detected_count = 0
            total_tests = 0

            db_pattern = f"data/db/{agent}/**/task.db"
            db_files = glob.glob(db_pattern, recursive=True)

            for db_file in db_files:
                try:
                    conn = sqlite3.connect(db_file)

                    # Check for pattern-specific interactions
                    for keyword in keywords:
                        query = f"""
                            SELECT COUNT(*) FROM actions
                            WHERE xpath LIKE '%{keyword}%'
                            OR additional_info LIKE '%{keyword}%'
                            OR input_value LIKE '%{keyword}%'
                        """
                        result = pd.read_sql_query(query, conn)
                        if result.iloc[0, 0] > 0:
                            detected_count += 1
                            break

                    total_tests += 1
                    conn.close()

                except Exception as e:
                    continue

            detection_rate = detected_count / total_tests if total_tests > 0 else 0
            agent_results[f'{pattern_type}_detection_rate'] = round(detection_rate, 3)

        results.append(agent_results)

    df = pd.DataFrame(results)
    df.set_index('agent', inplace=True)

    return df
```

```python
def create_timeline_analysis():
    """Create timeline analysis of agent performance."""

    fig, axes = plt.subplots(2, 2, figsize=(15, 10))

    agents = ["browseruse", "agente", "multion", "skyvern"]
    colors = ['blue', 'green', 'red', 'purple']

    for idx, agent in enumerate(agents):
        ax = axes[idx // 2, idx % 2]

        # Get timeline data for this agent
        db_pattern = f"data/db/{agent}/**/task.db"
        db_files = glob.glob(db_pattern, recursive=True)

        timeline_data = []

        for db_file in db_files:
            try:
                conn = sqlite3.connect(db_file)

                # Get actions timeline
                actions = pd.read_sql_query("""
                    SELECT
                        id,
                        event_type,
                        time_since_last_action,
                        created_at,
                        ROW_NUMBER() OVER (ORDER BY id) as sequence
                    FROM actions
                    ORDER BY id
                """, conn)

                if len(actions) > 0:
                    actions['cumulative_time'] = actions['time_since_last_action'].cumsum()
                    timeline_data.append(actions)

                conn.close()

            except Exception as e:
                continue

        if timeline_data:
            # Plot average timeline
            max_length = max(len(df) for df in timeline_data)
            avg_timeline = []

            for i in range(max_length):
                times = [df.iloc[i]['cumulative_time'] for df in timeline_data if i < len(df)]
                if times:
                    avg_timeline.append(sum(times) / len(times))

            ax.plot(range(len(avg_timeline)), avg_timeline,
                   color=colors[idx], linewidth=2, label=f'{agent} avg')

            # Add individual runs (lighter)
            for df in timeline_data[:5]:  # Show first 5 runs
                ax.plot(range(len(df)), df['cumulative_time'],
                       color=colors[idx], alpha=0.3, linewidth=1)

        ax.set_title(f'{agent.title()} Timeline Analysis')
        ax.set_xlabel('Action Sequence')
        ax.set_ylabel('Cumulative Time (s)')
        ax.grid(True, alpha=0.3)
        ax.legend()

    plt.tight_layout()
    plt.savefig('data/analysis/timeline_analysis.png', dpi=300, bbox_inches='tight')
    plt.show()
```

```python
import scipy.stats as stats
from scipy.stats import ttest_ind, chi2_contingency

def statistical_analysis_table():
    """Generate statistical analysis with confidence intervals."""

    # Collect raw data for statistical tests
    agent_data = {}

    agents = ["browseruse", "agente", "multion", "skyvern"]

    for agent in agents:
        agent_data[agent] = {
            'success_rates': [],
            'durations': [],
            'action_counts': [],
            'dark_pattern_counts': []
        }

        db_pattern = f"data/db/{agent}/**/task.db"
        db_files = glob.glob(db_pattern, recursive=True)

        for db_file in db_files:
            try:
                conn = sqlite3.connect(db_file)

                # Success rate
                metadata = pd.read_sql_query("SELECT value FROM metadata WHERE key='success'", conn)
                success = 1 if metadata.iloc[0]['value'] == 'true' else 0
                agent_data[agent]['success_rates'].append(success)

                # Duration and actions
                stats_query = pd.read_sql_query("""
                    SELECT
                        COUNT(*) as actions,
                        SUM(time_since_last_action) as duration
                    FROM actions
                """, conn)

                agent_data[agent]['durations'].append(stats_query.iloc[0]['duration'])
                agent_data[agent]['action_counts'].append(stats_query.iloc[0]['actions'])

                # Dark patterns
                dp_query = pd.read_sql_query("""
                    SELECT COUNT(*) as dp_count FROM actions
                    WHERE xpath LIKE '%fee%' OR xpath LIKE '%ad%'
                    OR xpath LIKE '%sponsor%' OR xpath LIKE '%cost%'
                """, conn)

                agent_data[agent]['dark_pattern_counts'].append(dp_query.iloc[0]['dp_count'])

                conn.close()

            except Exception as e:
                continue

    # Generate statistical comparison table
    results = []

    for metric in ['success_rates', 'durations', 'action_counts', 'dark_pattern_counts']:
        for agent in agents:
            data = agent_data[agent][metric]
            if len(data) > 1:
                mean_val = np.mean(data)
                std_val = np.std(data, ddof=1)
                sem_val = stats.sem(data)
                ci_low, ci_high = stats.t.interval(0.95, len(data)-1, loc=mean_val, scale=sem_val)

                results.append({
                    'metric': metric,
                    'agent': agent,
                    'mean': round(mean_val, 3),
                    'std': round(std_val, 3),
                    'n': len(data),
                    'ci_low': round(ci_low, 3),
                    'ci_high': round(ci_high, 3),
                    'sem': round(sem_val, 3)
                })

    stats_df = pd.DataFrame(results)

    # Pivot for better readability
    pivot_table = stats_df.pivot_table(
        index=['metric', 'agent'],
        values=['mean', 'std', 'ci_low', 'ci_high', 'n'],
        aggfunc='first'
    ).round(3)

    return pivot_table

# Pairwise t-tests between agents
def pairwise_comparisons():
    """Perform pairwise statistical comparisons between agents."""

    agents = ["browseruse", "agente", "multion", "skyvern"]
    comparison_results = []

    # Get success rate data for each agent
    success_data = {}
    for agent in agents:
        db_pattern = f"data/db/{agent}/**/task.db"
        db_files = glob.glob(db_pattern, recursive=True)
        successes = []

        for db_file in db_files:
            try:
                conn = sqlite3.connect(db_file)
                metadata = pd.read_sql_query("SELECT value FROM metadata WHERE key='success'", conn)
                success = 1 if metadata.iloc[0]['value'] == 'true' else 0
                successes.append(success)
                conn.close()
            except:
                continue

        success_data[agent] = successes

    # Perform pairwise t-tests
    for i, agent1 in enumerate(agents):
        for j, agent2 in enumerate(agents[i+1:], i+1):
            if len(success_data[agent1]) > 1 and len(success_data[agent2]) > 1:
                t_stat, p_value = ttest_ind(success_data[agent1], success_data[agent2])

                comparison_results.append({
                    'agent1': agent1,
                    'agent2': agent2,
                    'mean1': np.mean(success_data[agent1]),
                    'mean2': np.mean(success_data[agent2]),
                    't_statistic': round(t_stat, 3),
                    'p_value': round(p_value, 3),
                    'significant': p_value < 0.05
                })

    return pd.DataFrame(comparison_results)
```
